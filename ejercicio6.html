El archivo principal del microservicio (`main.py`) se mantuvo simple. El foco estuvo en el script de pruebas `locustfile.py`: 


# Pega tu fragmento de código aquí
# Ejemplo de un locustfile.py:

from locust import HttpUser, task, between

class QuickstartUser(HttpUser):
    wait_time = between(1, 2)

    @task
    def get_books(self):
        self.client.get("/books")

    @task(3) # Esta tarea se ejecutará 3 veces más que get_books
    def get_book_by_id(self):
        # Simula la obtención de un libro entre el ID 1 y 10
        book_id = 1 
        self.client.get(f"/books/{book_id}", name="/books/[id]")
